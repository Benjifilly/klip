- [ ] **Step 5: Implement `base64.ts`** (Hermes has no `btoa`/`Buffer`):

```ts
const ALPHA = 'ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/';
const REV = new Int8Array(128).fill(-1);
for (let i = 0; i < ALPHA.length; i++) REV[ALPHA.charCodeAt(i)] = i;

export function toBase64(bytes: Uint8Array): string {
  let out = '';
  for (let i = 0; i < bytes.length; i += 3) {
    const b0 = bytes[i], b1 = bytes[i + 1], b2 = bytes[i + 2];
    out += ALPHA[b0 >> 2] + ALPHA[((b0 & 3) << 4) | ((b1 ?? 0) >> 4)];
    out += b1 === undefined ? '=' : ALPHA[((b1 & 15) << 2) | ((b2 ?? 0) >> 6)];
    out += b2 === undefined ? '=' : ALPHA[b2 & 63];
  }
  return out;
}

export function fromBase64(b64: string): Uint8Array {
  const clean = b64.replace(/[^A-Za-z0-9+/]/g, '');
  const out = new Uint8Array(Math.floor((clean.length * 3) / 4));
  let o = 0;
  for (let i = 0; i + 1 < clean.length; i += 4) {
    const n = (REV[clean.charCodeAt(i)] << 18) | (REV[clean.charCodeAt(i + 1)] << 12) |
      ((REV[clean.charCodeAt(i + 2)] ?? 0) << 6) | (REV[clean.charCodeAt(i + 3)] ?? 0);
    out[o++] = n >> 16;
    if (i + 2 < clean.length) out[o++] = (n >> 8) & 255;
    if (i + 3 < clean.length) out[o++] = n & 255;
  }
  return out;
}
```

- [ ] **Step 6: Implement `crypto.ts`** — a line-for-line port of `client/electron/crypto.cjs` v2 (same salts, same Argon2id params, same emoji table, same bit-packing; the constants are the interop contract and MUST NOT be changed):

```ts
import { argon2id } from '@noble/hashes/argon2';
import { sha256 } from '@noble/hashes/sha256';
import { utf8ToBytes, bytesToHex } from '@noble/hashes/utils';
import { gcm } from '@noble/ciphers/aes';
import { getRandomValues } from './random';
import { toBase64, fromBase64 } from './base64';

const KEY_SALT = 'klip/v2/key';
const ROOM_SALT = 'klip/v2/room';
const FINGERPRINT_SALT = 'klip/v2/fingerprint';
const IV_BYTES = 12;
const ARGON2 = { t: 3, m: 64 * 1024, p: 1, dkLen: 32 };
const CODE_ALPHABET = 'abcdefghjkmnpqrstuvwxyz23456789';

export const FINGERPRINT_EMOJIS = [/* copy the exact 64-entry array from crypto.cjs */];

export interface Packet { iv: string; ct: string }

export function normalizeCode(code: unknown): string {
  return String(code ?? '').trim().toLowerCase().replace(/\s+/g, '');
}

export function formatSessionCode(raw: unknown): string | null {
  const code = normalizeCode(raw).replace(/-/g, '');
  if (!/^[a-z0-9]{16}$/.test(code)) return null;
  return `${code.slice(0, 4)}-${code.slice(4, 8)}-${code.slice(8, 12)}-${code.slice(12, 16)}`;
}

export function generateSessionCode(): string {
  const chars: string[] = [];
  while (chars.length < 16) {
    const buf = getRandomValues(new Uint8Array(32));
    for (const byte of buf) {
      if (byte >= 248) continue; // rejection sampling, no modulo bias
      chars.push(CODE_ALPHABET[byte % CODE_ALPHABET.length]);
      if (chars.length === 16) break;
    }
  }
  const s = chars.join('');
  return `${s.slice(0, 4)}-${s.slice(4, 8)}-${s.slice(8, 12)}-${s.slice(12, 16)}`;
}

/** Argon2id is synchronous in pure JS: yield to the UI first, then block. */
export async function deriveSessionKey(sessionCode: string): Promise<Uint8Array> {
  await new Promise((r) => setTimeout(r, 50));
  return argon2id(utf8ToBytes(normalizeCode(sessionCode)), utf8ToBytes(KEY_SALT), ARGON2);
}

export async function deriveRoomId(sessionCode: string): Promise<string> {
  return bytesToHex(sha256(utf8ToBytes(`${ROOM_SALT}:${normalizeCode(sessionCode)}`)));
}

export async function deriveFingerprint(sessionCode: string): Promise<string[]> {
  const d = sha256(utf8ToBytes(`${FINGERPRINT_SALT}:${normalizeCode(sessionCode)}`));
  const [b0, b1, b2] = d;
  const idx = [b0 >> 2, ((b0 & 3) << 4) | (b1 >> 4), ((b1 & 15) << 2) | (b2 >> 6), b2 & 63];
  return idx.map((i) => FINGERPRINT_EMOJIS[i]);
}

export async function encryptBytes(key: Uint8Array, bytes: Uint8Array): Promise<Packet> {
  const iv = getRandomValues(new Uint8Array(IV_BYTES));
  return { iv: toBase64(iv), ct: toBase64(gcm(key, iv).encrypt(bytes)) };
}

export async function decryptBytes(key: Uint8Array, packet: Packet): Promise<Uint8Array> {
  return gcm(key, fromBase64(packet.iv)).decrypt(fromBase64(packet.ct));
}

export async function encryptJSON(key: Uint8Array, value: unknown): Promise<Packet> {
  return encryptBytes(key, utf8ToBytes(JSON.stringify(value)));
}

export async function decryptJSON(key: Uint8Array, packet: Packet): Promise<unknown> {
  return JSON.parse(new TextDecoder().decode(await decryptBytes(key, packet)));
}
```

```ts
import { encryptBytes, decryptBytes, encryptJSON, decryptJSON, formatSessionCode, type Packet } from './crypto';
import { toBase64, fromBase64 } from './base64';
import { randomUUID } from './random';

const CLIP_FORMAT_VERSION = 1;
const CHUNK_BYTES = 256 * 1024;
const MAX_CHUNKS_PER_TRANSFER = 128;
const MAX_INFLIGHT_TRANSFERS = 4;
const MAX_INFLIGHT_CHUNK_BYTES = 32 * 1024 * 1024;
const MAX_IMAGE_BYTES = 4 * 1024 * 1024;
const PRESENCE_INTERVAL_MS = 60_000;
const PRESENCE_TTL_MS = 150_000;
const MAX_RECONNECT_DELAY_MS = 30_000;
const MID_MEMORY = 512;

export type Status = 'disconnected' | 'connecting' | 'connected';
export type IncomingClip =
  | { kind: 'text'; text: string; deviceName: string; ts: number }
  | { kind: 'image'; png: Uint8Array; deviceName: string; ts: number }
  | { kind: 'file'; name: string; deviceName: string; ts: number }; // placeholder — bytes dropped in V1

export interface PeerDevice { id: string; name: string; ts: number }

export interface RelayClientOptions {
  url: string; roomId: string; key: Uint8Array;
  installId: string; deviceName: string;
  onClip(clip: IncomingClip): void;
  onRotate(req: { code: string; deviceName: string }): void;
  onStatus(status: Status): void;
  onPeers(count: number): void;
  onPresence(devices: PeerDevice[]): void;
}

/** ws:// is only acceptable on loopback/private LAN — mirror of the desktop rule. */
export function isAllowedRelayUrl(raw: string): boolean {
  let u: { protocol: string; hostname: string };
  try { u = new URL(raw); } catch { return false; }
  if (u.protocol === 'wss:') return true;
  if (u.protocol !== 'ws:') return false;
  const h = u.hostname;
  return h === 'localhost' || h.endsWith('.local') ||
    /^127\./.test(h) || /^10\./.test(h) || /^192\.168\./.test(h) ||
    /^172\.(1[6-9]|2\d|3[01])\./.test(h);
}

interface Transfer { tot: number; parts: Map<number, Uint8Array>; bytes: number; ts: number }

export class RelayClient {
  private ws: WebSocket | null = null;
  private opts: RelayClientOptions;
  private seenMids = new Set<string>();
  private transfers = new Map<string, Transfer>();
  private inflightBytes = 0;
  private peers = new Map<string, { name: string; ts: number }>();
  private reconnectDelay = 1_000;
  private reconnectTimer: ReturnType<typeof setTimeout> | null = null;
  private presenceTimer: ReturnType<typeof setInterval> | null = null;
  private closed = true;

  constructor(opts: RelayClientOptions) { this.opts = opts; }

  connect(): void {
    this.closed = false;
    this.teardownSocket();
    this.opts.onStatus('connecting');
    const ws = new (globalThis as unknown as { WebSocket: typeof WebSocket }).WebSocket(this.opts.url);
    this.ws = ws;
    ws.addEventListener('open', () => {
      this.reconnectDelay = 1_000;
      ws.send(JSON.stringify({ type: 'join', roomId: this.opts.roomId }));
    });
    ws.addEventListener('message', (ev) => { this.onMessage(String(ev.data)).catch(() => {}); });
    ws.addEventListener('close', () => {
      this.opts.onStatus('disconnected');
      this.stopPresence();
      if (!this.closed) this.scheduleReconnect();
    });
    ws.addEventListener('error', () => { /* close follows */ });
  }

  disconnect(): void {
    this.closed = true;
    if (this.reconnectTimer) clearTimeout(this.reconnectTimer);
    this.reconnectTimer = null;
    this.sendPresence(false, true).catch(() => {}); // graceful bye
    this.stopPresence();
    this.teardownSocket();
    this.opts.onStatus('disconnected');
  }

  async sendText(text: string): Promise<void> {
    await this.sendEncrypted({ kind: 'text', text, deviceName: this.opts.deviceName, ts: Date.now() });
  }

  async sendImage(png: Uint8Array): Promise<void> {
    if (!png.length || png.length > MAX_IMAGE_BYTES) throw new Error('image too large (4 MiB max)');
    await this.sendEncrypted({ kind: 'image', png: toBase64(png), deviceName: this.opts.deviceName, ts: Date.now() });
  }

  private teardownSocket(): void {
    if (!this.ws) return;
    try { this.ws.close(); } catch { /* already closed */ }
    this.ws = null;
  }

  private scheduleReconnect(): void {
    if (this.reconnectTimer) return;
    this.reconnectTimer = setTimeout(() => { this.reconnectTimer = null; this.connect(); }, this.reconnectDelay);
    this.reconnectDelay = Math.min(this.reconnectDelay * 2, MAX_RECONNECT_DELAY_MS);
  }

  private rememberMid(mid: unknown): boolean {
    if (typeof mid !== 'string' || !mid) return false;
    if (this.seenMids.has(mid)) return true;
    this.seenMids.add(mid);
    if (this.seenMids.size > MID_MEMORY) {
      const oldest = this.seenMids.values().next().value as string;
      this.seenMids.delete(oldest);
    }
    return false;
  }

  private async onMessage(raw: string): Promise<void> {
    let msg: { type?: string; count?: number; sid?: string; mid?: string; data?: Packet & { fid?: string; seq?: number; tot?: number } };
    try { msg = JSON.parse(raw); } catch { return; }
    switch (msg.type) {
      case 'joined':
        this.opts.onStatus('connected');
        this.startPresence();
        this.sendPresence(true).catch(() => {});
        return;
      case 'peers':
        this.opts.onPeers(Number(msg.count) || 0);
        this.prunePeers();
        return;
      case 'clip': {
        if (!msg.data) return;
        if (msg.sid && msg.sid === this.opts.installId) return;
        if (this.rememberMid(msg.mid)) return;
        if (msg.data.fid !== undefined) return this.onChunk(msg.data as Packet & { fid: string; seq: number; tot: number });
        let item: unknown;
        try { item = await decryptJSON(this.opts.key, msg.data); } catch { return; }
        this.applyIncoming(item);
        return;
      }
      default: return; // hello, left, error, pong
    }
  }

  private async onChunk(data: Packet & { fid: string; seq: number; tot: number }): Promise<void> {
    if (typeof data.fid !== 'string' || data.fid.length > 64 ||
      !Number.isInteger(data.seq) || !Number.isInteger(data.tot) ||
      data.seq < 0 || data.tot < 1 || data.tot > MAX_CHUNKS_PER_TRANSFER || data.seq >= data.tot) return;
    let bytes: Uint8Array;
    try { bytes = await decryptBytes(this.opts.key, data); } catch { return; }
    let transfer = this.transfers.get(data.fid);
    if (!transfer) {
      if (this.transfers.size >= MAX_INFLIGHT_TRANSFERS) return;
      transfer = { tot: data.tot, parts: new Map(), bytes: 0, ts: Date.now() };
      this.transfers.set(data.fid, transfer);
    }
    if (data.tot !== transfer.tot || transfer.parts.has(data.seq)) return;
    transfer.parts.set(data.seq, bytes);
    transfer.bytes += bytes.length;
    this.inflightBytes += bytes.length;
    if (this.inflightBytes > MAX_INFLIGHT_CHUNK_BYTES) return this.dropTransfer(data.fid);
    if (transfer.parts.size < transfer.tot) return;
    this.dropTransfer(data.fid);
    let total = 0;
    for (let s = 0; s < transfer.tot; s++) total += transfer.parts.get(s)?.length ?? 0;
    const joined = new Uint8Array(total);
    let off = 0;
    for (let s = 0; s < transfer.tot; s++) {
      const part = transfer.parts.get(s);
      if (!part) return;
      joined.set(part, off); off += part.length;
    }
    let item: unknown;
    try { item = JSON.parse(new TextDecoder().decode(joined)); } catch { return; }
    this.applyIncoming(item);
  }

  private dropTransfer(fid: string): void {
    const t = this.transfers.get(fid);
    if (!t) return;
    this.inflightBytes -= t.bytes;
    this.transfers.delete(fid);
  }

  private applyIncoming(raw: unknown): void {
    if (!raw || typeof raw !== 'object') return;
    const item = raw as Record<string, unknown>;
    if (Number.isInteger(item.v) && (item.v as number) > CLIP_FORMAT_VERSION) return;
    const deviceName = String(item.deviceName ?? 'Unknown device').slice(0, 64);
    const ts = typeof item.ts === 'number' ? item.ts : Date.now();
    if (item.kind === 'presence') {
      const id = String(item.id ?? '').slice(0, 64);
      if (!id || id === this.opts.installId) return;
      if (item.bye) { this.peers.delete(id); this.emitPresence(); return; }
      const prev = this.peers.get(id);
      this.peers.set(id, { name: deviceName, ts: Date.now() });
      if (item.hello) this.sendPresence(false).catch(() => {});
      if (!prev || prev.name !== deviceName) this.emitPresence();
      return;
    }
    if (item.kind === 'rotate') {
      const code = formatSessionCode(item.code);
      if (!code) return;
      this.opts.onRotate({ code, deviceName }); // consent is the UI's job — never follow silently
      return;
    }
    if (item.kind === 'image') {
      if (typeof item.png !== 'string') return;
      const png = fromBase64(item.png);
      if (!png.length || png.length > MAX_IMAGE_BYTES) return;
      this.opts.onClip({ kind: 'image', png, deviceName, ts });
      return;
    }
    if (item.kind === 'file') {
      const name = String(item.name ?? 'file').slice(0, 128);
      this.opts.onClip({ kind: 'file', name, deviceName, ts }); // V1: placeholder only
      return;
    }
    if (typeof item.text === 'string' && item.text) {
      this.opts.onClip({ kind: 'text', text: item.text, deviceName, ts });
    }
  }

  private async sendEncrypted(payload: Record<string, unknown>): Promise<void> {
    const ws = this.ws;
    if (!ws || ws.readyState !== 1 /* OPEN */) return;
    const versioned = { v: CLIP_FORMAT_VERSION, ...payload };
    const mid = randomUUID();
    const bytes = new TextEncoder().encode(JSON.stringify(versioned));
    if (bytes.length <= CHUNK_BYTES) {
      this.rememberMid(mid);
      const data = await encryptJSON(this.opts.key, versioned);
      ws.send(JSON.stringify({ type: 'clip', v: CLIP_FORMAT_VERSION, mid, sid: this.opts.installId, data }));
      return;
    }
    const tot = Math.ceil(bytes.length / CHUNK_BYTES);
    if (tot > MAX_CHUNKS_PER_TRANSFER) throw new Error('payload too large');
    const fid = randomUUID();
    for (let seq = 0; seq < tot; seq++) {
      const slice = bytes.subarray(seq * CHUNK_BYTES, (seq + 1) * CHUNK_BYTES);
      const data = await encryptBytes(this.opts.key, slice);
      const chunkMid = `${mid}:${seq}`;
      this.rememberMid(chunkMid);
      if (!this.ws || this.ws.readyState !== 1) return;
      this.ws.send(JSON.stringify({ type: 'clip', v: CLIP_FORMAT_VERSION, mid: chunkMid, sid: this.opts.installId, data: { ...data, fid, seq, tot } }));
    }
  }

  private async sendPresence(hello: boolean, bye = false): Promise<void> {
    await this.sendEncrypted({
      kind: 'presence', id: this.opts.installId, name: this.opts.deviceName,
      hello, bye, ts: Date.now(),
    });
  }

  private startPresence(): void {
    this.stopPresence();
    this.presenceTimer = setInterval(() => {
      this.sendPresence(false).catch(() => {});
      this.prunePeers();
    }, PRESENCE_INTERVAL_MS);
  }

  private stopPresence(): void {
    if (this.presenceTimer) clearInterval(this.presenceTimer);
    this.presenceTimer = null;
  }

  private prunePeers(): void {
    const cutoff = Date.now() - PRESENCE_TTL_MS;
    let changed = false;
    for (const [id, p] of this.peers) if (p.ts < cutoff) { this.peers.delete(id); changed = true; }
    if (changed) this.emitPresence();
  }

  private emitPresence(): void {
    this.opts.onPresence([...this.peers].map(([id, p]) => ({ id, name: p.name, ts: p.ts })));
  }
}
```

```ts
export const MAX_ITEMS = 200;
export const MAX_IMAGE_BLOBS = 20;

export interface HistoryItem {
  id: string;
  kind: 'text' | 'image' | 'file-placeholder';
  text?: string;
  blobId?: string;     // image bytes live in encrypted blob storage, not in the JSON
  fileName?: string;
  deviceName: string;
  direction: 'sent' | 'received';
  ts: number;
  pinned: boolean;
}

export function evict(items: HistoryItem[]): { kept: HistoryItem[]; deletedBlobIds: string[] } {
  const deletedBlobIds: string[] = [];
  let kept = [...items];
  // list cap: drop oldest unpinned first
  while (kept.length > MAX_ITEMS) {
    const idx = kept.map((it, i) => ({ it, i })).reverse().find(({ it }) => !it.pinned)?.i;
    if (idx === undefined) break;
    const [dropped] = kept.splice(idx, 1);
    if (dropped.blobId) deletedBlobIds.push(dropped.blobId);
  }
  // image-blob cap: drop oldest unpinned images beyond the budget
  const images = kept.filter((it) => it.kind === 'image' && !it.pinned);
  for (let i = 0; i < images.length - MAX_IMAGE_BLOBS + kept.filter((it) => it.kind === 'image' && it.pinned).length; i++) {
    const victim = images[images.length - 1 - i];
    if (!victim) break;
    kept = kept.filter((it) => it.id !== victim.id);
    if (victim.blobId) deletedBlobIds.push(victim.blobId);
  }
  return { kept, deletedBlobIds };
}

export function addItem(items: HistoryItem[], item: HistoryItem): HistoryItem[] {
  return evict([item, ...items]).kept;
}
```

```ts
import { AppState } from 'react-native';
import { RelayClient, isAllowedRelayUrl, type Status, type IncomingClip, type PeerDevice } from './protocol';
import { deriveSessionKey, deriveRoomId, deriveFingerprint, formatSessionCode } from './crypto';
import { addItem, type HistoryItem } from './history';
import * as Clipboard from 'expo-clipboard';
import { randomUUID } from './random';
import { session, blobs, historyFile, prefs, getInstallId } from './storage';
import { toBase64 } from './base64';

export const DEFAULT_RELAY = 'wss://klip-relay.fly.dev'; // match desktop default (check client settings default)

interface Prefs { relayUrl: string; deviceName: string; autoCopy: 'all' | 'text' | 'off' }

interface KlipState {
  hydrated: boolean;
  status: Status;
  code: string | null;
  fingerprint: string[];
  peerCount: number;
  devices: PeerDevice[];
  history: HistoryItem[];
  prefs: Prefs;
  pendingRotation: { code: string; deviceName: string } | null;
  lastDeriveMs: number | null;
  hydrate(): Promise<void>;
  join(code: string, relayUrl?: string): Promise<void>;
  leave(): Promise<void>;
  sendClipboard(): Promise<'text' | 'image' | 'empty'>;
  sendImage(png: Uint8Array): Promise<void>;
  copyItem(item: HistoryItem): Promise<void>;
  togglePin(id: string): void;
  removeItem(id: string): void;
  setPrefs(p: Partial<Prefs>): void;
  acceptRotation(): Promise<void>;
  dismissRotation(): void;
}

let client: RelayClient | null = null;

export const useKlip = create<KlipState>((set, get) => {
  async function persistHistory() { await historyFile.save(get().history).catch(() => {}); }

  async function onClip(clip: IncomingClip) {
    const { prefs } = get();
    let item: HistoryItem;
    if (clip.kind === 'image') {
      const blobId = await blobs.save(clip.png);
      item = { id: randomUUID(), kind: 'image', blobId, deviceName: clip.deviceName, direction: 'received', ts: clip.ts, pinned: false };
      if (prefs.autoCopy === 'all') await Clipboard.setImageAsync(toBase64(clip.png)).catch(() => {});
    } else if (clip.kind === 'file') {
      item = { id: randomUUID(), kind: 'file-placeholder', fileName: clip.name, deviceName: clip.deviceName, direction: 'received', ts: clip.ts, pinned: false };
    } else {
      item = { id: randomUUID(), kind: 'text', text: clip.text, deviceName: clip.deviceName, direction: 'received', ts: clip.ts, pinned: false };
      if (prefs.autoCopy !== 'off') await Clipboard.setStringAsync(clip.text).catch(() => {});
    }
    set((s) => ({ history: addItem(s.history, item) }));
    await persistHistory();
  }

  function buildClient(relayUrl: string, roomId: string, key: Uint8Array, installId: string) {
    client?.disconnect();
    client = new RelayClient({
      url: relayUrl, roomId, key, installId, deviceName: get().prefs.deviceName,
      onClip: (c) => { onClip(c).catch(() => {}); },
      onRotate: (r) => set({ pendingRotation: r }),
      onStatus: (status) => set({ status }),
      onPeers: (peerCount) => set({ peerCount }),
      onPresence: (devices) => set({ devices }),
    });
    client.connect();
  }

  AppState.addEventListener('change', (state) => {
    if (state === 'active' && get().code && get().status === 'disconnected') client?.connect();
  });

  return {
    hydrated: false, status: 'disconnected', code: null, fingerprint: [],
    peerCount: 0, devices: [], history: [],
    prefs: { relayUrl: DEFAULT_RELAY, deviceName: 'iPhone', autoCopy: 'text' },
    pendingRotation: null, lastDeriveMs: null,

    async hydrate() {
      const [saved, storedPrefs, storedHistory, installId] = await Promise.all([
        session.load(), prefs.load(), historyFile.load(), getInstallId(),
      ]);
      if (storedPrefs) set({ prefs: { ...get().prefs, ...(storedPrefs as Partial<Prefs>) } });
      if (Array.isArray(storedHistory)) set({ history: storedHistory as HistoryItem[] });
      if (saved) {
        const [roomId, fingerprint] = await Promise.all([deriveRoomId(saved.code), deriveFingerprint(saved.code)]);
        set({ code: saved.code, fingerprint });
        buildClient(get().prefs.relayUrl, roomId, saved.key, installId);
      }
      set({ hydrated: true });
    },

    async join(rawCode, relayUrl) {
      const code = formatSessionCode(rawCode);
      if (!code) throw new Error('Not a Klip code');
      const url = relayUrl ?? get().prefs.relayUrl;
      if (!isAllowedRelayUrl(url)) throw new Error('Relay must be wss:// (or a LAN ws:// address)');
      const t0 = Date.now();
      const key = await deriveSessionKey(code); // slow: pure-JS Argon2id, only at pairing
      set({ lastDeriveMs: Date.now() - t0 });
      const [roomId, fingerprint] = await Promise.all([deriveRoomId(code), deriveFingerprint(code)]);
      await session.save(code, key);
      if (relayUrl) get().setPrefs({ relayUrl });
      set({ code, fingerprint });
      buildClient(url, roomId, key, await getInstallId());
    },

    async leave() {
      client?.disconnect(); client = null;
      await session.clear();
      for (const it of get().history) if (it.blobId) await blobs.remove(it.blobId);
      set({ code: null, fingerprint: [], history: [], peerCount: 0, devices: [], status: 'disconnected' });
      await historyFile.save([]);
    },

    async sendClipboard() {
      if (await Clipboard.hasImageAsync()) {
        const img = await Clipboard.getImageAsync({ format: 'png' });
        if (img?.data) {
          const b64 = img.data.replace(/^data:image\/\w+;base64,/, '');
          const { fromBase64 } = await import('./base64');
          const png = fromBase64(b64);
          await get().sendImage(png);
          return 'image';
        }
      }
      const text = await Clipboard.getStringAsync();
      if (!text) return 'empty';
      await client?.sendText(text);
      const item: HistoryItem = { id: randomUUID(), kind: 'text', text, deviceName: get().prefs.deviceName, direction: 'sent', ts: Date.now(), pinned: false };
      set((s) => ({ history: addItem(s.history, item) }));
      await persistHistory();
      return 'text';
    },

    async sendImage(png) {
      await client?.sendImage(png);
      const blobId = await blobs.save(png);
      const item: HistoryItem = { id: randomUUID(), kind: 'image', blobId, deviceName: get().prefs.deviceName, direction: 'sent', ts: Date.now(), pinned: false };
      set((s) => ({ history: addItem(s.history, item) }));
      await persistHistory();
    },

    async copyItem(item) {
      if (item.kind === 'text' && item.text) await Clipboard.setStringAsync(item.text);
      if (item.kind === 'image' && item.blobId) {
        const png = await blobs.load(item.blobId);
        if (png) await Clipboard.setImageAsync(toBase64(png));
      }
    },

    togglePin(id) {
      set((s) => ({ history: s.history.map((it) => (it.id === id ? { ...it, pinned: !it.pinned } : it)) }));
      persistHistory();
    },

    removeItem(id) {
      const victim = get().history.find((it) => it.id === id);
      if (victim?.blobId) blobs.remove(victim.blobId);
      set((s) => ({ history: s.history.filter((it) => it.id !== id) }));
      persistHistory();
    },

    setPrefs(p) {
```

```ts
import { formatSessionCode } from './crypto';
import { isAllowedRelayUrl } from './protocol';

export function parseJoinLink(raw: string): { code: string; relay: string | null } | null {
  const m = /^klip:\/\/join\/?\?(.+)$/i.exec(String(raw ?? '').trim());
  if (!m) return null;
  const params = new Map<string, string>();
  for (const pair of m[1].split('&')) {
    const eq = pair.indexOf('=');
    if (eq <= 0) continue;
    try { params.set(decodeURIComponent(pair.slice(0, eq)), decodeURIComponent(pair.slice(eq + 1))); } catch { /* skip bad pair */ }
  }
  const code = formatSessionCode(params.get('code'));
  if (!code) return null;
  const relayRaw = (params.get('relay') ?? '').trim();
  const relay = relayRaw && /^wss?:\/\/.+/.test(relayRaw) && isAllowedRelayUrl(relayRaw) ? relayRaw : null;
  return { code, relay };
}
```

```tsx
            <Text className="font-mono text-klip-300">{reveal ? code : '••••-••••-••••-••••'}</Text>
            <Pressable hitSlop={8} onPress={() => setReveal(!reveal)}><Text className="text-zinc-500">{reveal ? 'Hide' : 'Show'}</Text></Pressable>
          </View>
          <Fingerprint emojis={fingerprint} />
          <Pressable
            className="rounded-xl border border-red-400/40 px-4 py-3"
            onPress={() =>
              Alert.alert('Leave session?', 'Your history on this phone is deleted.', [
                { text: 'Cancel', style: 'cancel' },
                { text: 'Leave', style: 'destructive', onPress: async () => { await leave(); router.replace('/pairing'); } },
              ])
            }>
            <Text className="text-center text-red-400">Leave session</Text>
          </Pressable>
        </View>
      )}

      <Text className="mt-8 text-xs text-zinc-600">
        Klip iOS {Constants.expoConfig?.version ?? 'dev'} · protocol v1
        {lastDeriveMs ? ` · key derivation ${(lastDeriveMs / 1000).toFixed(1)}s on this device` : ''}
      </Text>
    </ScrollView>
  );
}
```
